  message(FATAL_ERROR "Set NODE_INCLUDE_DIR to the directory containing node_api.h")

  if(MINGW)

    target_link_libraries(multileveldb_node PRIVATE ${NODE_LIBRARY})

  add_custom_command(TARGET multileveldb_node POST_BUILD COMMAND ${CMAKE_COMMAND} -E
