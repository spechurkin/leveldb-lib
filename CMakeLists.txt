cmake_minimum_required(VERSION 3.20)
project(multileveldb VERSION 1.0.0 HOMEPAGE_URL "https://github.com/spechurkin/leveldb-lib" LANGUAGES C CXX)
if(NOT DEFINED CMAKE_C_STANDARD)
  set(CMAKE_C_STANDARD 11)
endif()
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_definitions(_DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
endif()
include(CTest)
include(GNUInstallDirs)
include(FetchContent)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SNAPPY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SNAPPY_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SNAPPY_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(snappy GIT_REPOSITORY https://github.com/google/snappy.git
  GIT_TAG 9c28114a38866f6deeaa826db918293bc28ae410 GIT_SUBMODULES "")
FetchContent_MakeAvailable(snappy)
set_target_properties(snappy PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set(HAVE_SNAPPY 1 CACHE INTERNAL "Bundled Snappy is available" FORCE)
set(HAVE_CRC32C 0 CACHE INTERNAL "Use the engine's portable CRC implementation" FORCE)
set(HAVE_TCMALLOC 0 CACHE INTERNAL "Do not add an ambient allocator dependency" FORCE)
FetchContent_Declare(leveldb GIT_REPOSITORY https://github.com/google/leveldb.git
  GIT_TAG 99b3c03b3284f5886f9ef9a4ef703d57373e61be GIT_SUBMODULES "")
FetchContent_MakeAvailable(leveldb)
set_target_properties(leveldb PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set_target_properties(leveldbutil PROPERTIES EXCLUDE_FROM_ALL YES)
add_subdirectory(native)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME process_lock COMMAND ${CMAKE_COMMAND} -E env
      "MULTILEVELDB_LIBRARY=$<TARGET_FILE:multileveldb>" "ML_TEST_ROOT=${CMAKE_BINARY_DIR}"
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/abi/process_lock.py)
    add_test(NAME python_binding COMMAND ${CMAKE_COMMAND} -E env
      "MULTILEVELDB_LIBRARY=$<TARGET_FILE:multileveldb>" "ML_TEST_ROOT=${CMAKE_BINARY_DIR}"
      "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/bindings/python"
      ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/bindings/python/tests -v)
  endif()
  add_executable(interop_c tests/integration/verify.c)
  target_link_libraries(interop_c PRIVATE multileveldb)
  add_executable(abi_contract tests/abi/contract.cpp)
  target_link_libraries(abi_contract PRIVATE multileveldb)
  add_test(NAME abi_contract COMMAND abi_contract "${CMAKE_BINARY_DIR}/abi")
  add_executable(c_integration tests/native/integration.c
          native/include/multileveldb.h)
  target_link_libraries(c_integration PRIVATE multileveldb)
  add_test(NAME c_integration COMMAND c_integration "${CMAKE_BINARY_DIR}/test-c")
  add_executable(cpp_integration tests/native/integration.cpp
          native/include/multileveldb.h)
  target_link_libraries(cpp_integration PRIVATE multileveldb leveldb)
  add_test(NAME cpp_integration COMMAND cpp_integration "${CMAKE_BINARY_DIR}/test-cpp")
  if(MSVC)
    target_compile_options(interop_c PRIVATE /UNDEBUG)
    target_compile_options(abi_contract PRIVATE /UNDEBUG)
    target_compile_options(c_integration PRIVATE /UNDEBUG)
    target_compile_options(cpp_integration PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(interop_c PRIVATE -UNDEBUG)
    target_compile_options(abi_contract PRIVATE -UNDEBUG)
    target_compile_options(c_integration PRIVATE -UNDEBUG)
    target_compile_options(cpp_integration PRIVATE -UNDEBUG)
  endif()
  if(WIN32 AND CMAKE_VERSION VERSION_GREATER_EQUAL 3.22)
    get_property(ml_native_tests DIRECTORY PROPERTY TESTS)
    set_property(TEST ${ml_native_tests} PROPERTY ENVIRONMENT_MODIFICATION
      "PATH=path_list_prepend:$<TARGET_FILE_DIR:multileveldb>")
  endif()
endif()
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/multileveldb)
install(FILES ${leveldb_SOURCE_DIR}/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/multileveldb RENAME LICENSE.leveldb)
install(FILES ${snappy_SOURCE_DIR}/COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/multileveldb RENAME LICENSE.snappy)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_NAME multileveldb)
set(CPACK_GENERATOR TGZ ZIP)
include(CPack)
