cmake_minimum_required(VERSION 3.20)
project(multileveldb_consumer LANGUAGES C CXX)
include(CTest)
find_package(multileveldb CONFIG REQUIRED)
add_executable(consumer_c ../../examples/c/main.c)
target_link_libraries(consumer_c PRIVATE multileveldb::multileveldb)
add_executable(consumer_cpp ../../examples/cpp/main.cpp)
target_link_libraries(consumer_cpp PRIVATE multileveldb::multileveldb)
add_test(NAME consumer_c COMMAND consumer_c)
add_test(NAME consumer_cpp COMMAND consumer_cpp)
set_tests_properties(consumer_c consumer_cpp PROPERTIES RUN_SERIAL TRUE)
